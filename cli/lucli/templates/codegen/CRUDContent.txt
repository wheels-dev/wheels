|DescriptionComment|component extends="Controller" {

	/**
	 * Controller config. `super.config()` inherits CSRF protection
	 * (`protectsFromForgery()`) from app/controllers/Controller.cfc. Keep it,
	 * and keep the call when you add your own filters — without it `startFormTag`
	 * silently stops emitting the hidden `authenticityToken` input.
	 */
	function config() {
		super.config();
		// 404 rather than a 500 when the key matches nothing. findByKey returns a
		// non-object for BOTH a missing key and a soft-deleted row, and the
		// actions below then operate on that empty value — `post.title` in the
		// view, or `.update()` here, throws.
		//
		// A BEFORE filter that deliberately does NOT assign the record: the
		// actions keep their conventional single-finder body, which the
		// `--belongsTo` parent wiring depends on — it only rewrites `show()`
		// when the finder is the whole body, and skips a customized one.
		// Loading twice is cheap and keeps that contract intact.
		filters(through="requireRecord", only="show,edit,update,delete");
	}

	/**
	 * Actions set view-facing variables UNscoped (e.g. `|ObjectNamePlural|=…`),
	 * which lands them in the controller's `variables` scope so the rendered
	 * view can read them. Do NOT switch these to `local.` — the view can't see
	 * `local.`, and its query loop would throw "Can't cast String to query".
	 * Reserve `local.` for temporaries the view doesn't need.
	 */
	/**
	* View all |ObjectNamePluralC|
	**/
	function index() {
		|ObjectNamePlural|=model("|ObjectNameSingularC|").findAll();
	}

	/**
	* View |ObjectNameSingularC|
	**/
	function show() {
		|ObjectNameSingular|=model("|ObjectNameSingularC|").findByKey(params.key);
	}

	/**
	* Add New |ObjectNameSingularC|
	**/
	function new() {
		|ObjectNameSingular|=model("|ObjectNameSingularC|").new();
	}

	/**
	* Create |ObjectNameSingularC|
	**/
	function create() {
		|ObjectNameSingular|=model("|ObjectNameSingularC|").new(params.|ObjectNameSingular|);
		if(|ObjectNameSingular|.save()){
			flashInsert(success="|ObjectNameSingularC| was created successfully.");
			redirectTo(route="|ObjectNameSingular|", key=|ObjectNameSingular|.id);
		} else {
			renderView(action="new");
		}
	}

	/**
	* Edit |ObjectNameSingularC|
	**/
	function edit() {
		|ObjectNameSingular|=model("|ObjectNameSingularC|").findByKey(params.key);
	}

	/**
	* Update |ObjectNameSingularC|
	**/
	function update() {
		|ObjectNameSingular|=model("|ObjectNameSingularC|").findByKey(params.key);
		if(|ObjectNameSingular|.update(params.|ObjectNameSingular|)){
			flashInsert(success="|ObjectNameSingularC| was updated successfully.");
			redirectTo(route="|ObjectNameSingular|", key=|ObjectNameSingular|.id);
		} else {
			renderView(action="edit");
		}
	}

	/**
	* Delete |ObjectNameSingularC|
	**/
	function delete() {
		|ObjectNameSingular|=model("|ObjectNameSingularC|").findByKey(params.key);
		|ObjectNameSingular|.delete();
		flashInsert(success="|ObjectNameSingularC| was deleted successfully.");
		redirectTo(route="|ObjectNamePlural|");
	}

	/**
	* Guard: abort with a 404 when the key matches no record.
	*
	* The message is a plain string on purpose. The scaffold's parent wiring
	* scans this file with ScaffoldSource, which refuses interpolated strings
	* (`##...##`) wholesale — a single `##params.key##` here would silently turn
	* off `include=` wiring for every `--belongsTo` child of this model.
	**/
	private function requireRecord() {
		if (!IsObject(model("|ObjectNameSingularC|").findByKey(key=params.key))) {
			Throw(
				type = "Wheels.RecordNotFound",
				message = "|ObjectNameSingularC| not found for the requested key."
			);
		}
	}

}
